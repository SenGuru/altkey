from openai import OpenAI

client = OpenAI(
    base_url="https://api.altkey.dev/v1",
    api_key="ak-live-7f3c4b8a920e4f17ad92a",
)

stream = client.chat.completions.create(
    model="claude-3-5-sonnet",   # or "gpt-4o"
    messages=[{"role": "user", "content": "Summarize my repo."}],
    stream=True,
)
for chunk in stream:
    print(chunk.choices[0].delta.content or "", end="")
